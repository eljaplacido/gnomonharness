import json, os, sys, math, random, argparse, time
import torch
from torch.utils.data import DataLoader
from transformers import AutoTokenizer, AutoModelForCausalLM, get_linear_schedule_with_warmup
from peft import LoraConfig, get_peft_model
torch.set_num_threads(20)
BASE='openai-community/gpt2'
letters=['A','B','C','D']

def make_prompt(s, t):
    q=s['question']; opts=s['options']
    if t=='canon':
        return f"Question: {q}\n"+"\n".join(opts)+"\nAnswer:"
    if t=='options':
        return f"Question: {q}\nOptions:\n"+"\n".join(opts)+"\nAnswer:"
    if t=='plain':
        return f"{q}\n"+"\n".join(opts)+"\nAnswer:"
    if t=='theanswer':
        return f"Question: {q}\n"+"\n".join(opts)+"\nThe answer is"
    if t=='canon_sp':
        return f"Question: {q}\n"+"\n".join(opts)+"\nAnswer: "
    if t=='q_only':
        return f"Question: {q}\nAnswer:"
    raise ValueError(t)

class DS(torch.utils.data.Dataset):
    def __init__(self, data, templates, add_eos=False):
        self.items=[]
        for s in data:
            for t in templates:
                p=make_prompt(s,t)
                a=(' '+s['answer']) if not p.endswith(' ') else s['answer']
                self.items.append((p,a,add_eos))
    def __len__(self): return len(self.items)
    def __getitem__(self,i): return self.items[i]

def collate(batch, tok, maxlen=256):
    texts=[p+a+((tok.eos_token or '') if e else '') for p,a,e in batch]
    prompts=[p for p,a,e in batch]
    ans=[a for p,a,e in batch]
    enc=tok(texts, return_tensors='pt', padding=True, truncation=True, max_length=maxlen)
    labels=torch.full_like(enc.input_ids, -100)
    for i,(p,a,e) in enumerate(batch):
        pl=len(tok(p, add_special_tokens=False).input_ids)
        # with special tokens, gpt2 adds none; assume pl valid
        labels[i,pl]=tok(a, add_special_tokens=False).input_ids[0]
        if e and pl+1<labels.shape[1]:
            labels[i,pl+1]=tok.eos_token_id
    return enc, labels

def eval_acc(model, tok, data, template='canon', bs=50, include_sp=True):
    model.eval(); correct=0
    for i in range(0,len(data),bs):
        chunk=data[i:i+bs]
        texts=[make_prompt(s,template) for s in chunk]
        enc=tok(texts, return_tensors='pt', padding=True)
        with torch.no_grad(): logits=model(**enc).logits
        lens=enc.attention_mask.sum(1)-1
        for j,s in enumerate(chunk):
            lg=logits[j,lens[j]]
            scores=[]
            for L in letters:
                sc=lg[tok.encode(' '+L, add_special_tokens=False)[0]].item()
                if include_sp:
                    sc=max(sc,lg[tok.encode(L, add_special_tokens=False)[0]].item())
                scores.append(sc)
            pred=letters[max(range(4),key=lambda k:scores[k])]
            correct+=pred==s['answer']
    model.train()
    return correct/len(data)

def main():
    ap=argparse.ArgumentParser()
    ap.add_argument('--templates', default='canon')
    ap.add_argument('--epochs', type=int, default=10)
    ap.add_argument('--lr', type=float, default=5e-4)
    ap.add_argument('--bs', type=int, default=16)
    ap.add_argument('--dropout', type=float, default=0.05)
    ap.add_argument('--add_eos', action='store_true')
    ap.add_argument('--seed', type=int, default=42)
    ap.add_argument('--out', default='/tmp/expout')
    args=ap.parse_args()
    random.seed(args.seed); torch.manual_seed(args.seed)
    train=json.load(open('/app/data/train.json')); val=json.load(open('/app/data/val.json'))
    templates=args.templates.split(',')
    tok=AutoTokenizer.from_pretrained(BASE); tok.pad_token=tok.eos_token
    model=AutoModelForCausalLM.from_pretrained(BASE)
    model.config.pad_token_id=tok.pad_token_id
    lc=LoraConfig(r=8,lora_alpha=16,target_modules=['c_attn'],lora_dropout=args.dropout,bias='none',task_type='CAUSAL_LM')
    model=get_peft_model(model,lc)
    model.print_trainable_parameters()
    ds=DS(train,templates,args.add_eos)
    dl=DataLoader(ds,batch_size=args.bs,shuffle=True,collate_fn=lambda b:collate(b,tok))
    opt=torch.optim.AdamW([p for p in model.parameters() if p.requires_grad],lr=args.lr,weight_decay=0.01)
    total=len(dl)*args.epochs
    sched=get_linear_schedule_with_warmup(opt,int(total*0.05),total)
    print('examples',len(ds),'steps',total,flush=True)
    for ep in range(1,args.epochs+1):
        t0=time.time(); model.train(); losses=[]
        for enc,labels in dl:
            out=model(**enc,labels=labels)
            loss=out.loss
            loss.backward(); torch.nn.utils.clip_grad_norm_([p for p in model.parameters() if p.requires_grad],1.0)
            opt.step(); sched.step(); opt.zero_grad()
            losses.append(loss.item())
        print(f"ep {ep} loss {sum(losses)/len(losses):.4f} time {time.time()-t0:.1f}",flush=True)
        for t in ['canon','options','plain','theanswer','canon_sp']:
            print(' ',t,eval_acc(model,tok,val,t),flush=True)
        os.makedirs(args.out,exist_ok=True)
        model.save_pretrained(args.out); tok.save_pretrained(args.out)
if __name__=='__main__': main()
