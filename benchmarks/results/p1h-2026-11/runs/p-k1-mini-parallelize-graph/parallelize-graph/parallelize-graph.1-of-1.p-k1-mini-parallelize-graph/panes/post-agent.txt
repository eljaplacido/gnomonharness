#define MAX_KMER_LEN 64
#define KM_STRIDE (MAX_KMER_LEN + 1)
#define TABLE_SIZE 131072
#define TABLE_MASK (TABLE_SIZE - 1)

shared char kmer_data[MAX_N * KM_STRIDE];
shared char left_ext[MAX_N];
shared char right_ext[MAX_N];

shared int buckets[TABLE_SIZE];
shared int hash_next[MAX_N];
shared int graph_next[MAX_N];
shared int succ_bucket[MAX_N];
shared int indeg[MAX_N];
shared int outdeg[MAX_N];
shared int visited[MAX_N];

shared long g_n;
shared int g_k;
shared int g_error;

static uint64_t hash_local(const char *s) {
    uint64_t h = 1469598103934665603ULL;
    while (*s) {
        h ^= (unsigned char)(*s++);
        h *= 1099511628211ULL;
    }
    return h;
}

static uint64_t hash_shared_kmer(long idx, int k) {
    uint64_t h = 1469598103934665603ULL;
    long base = idx * KM_STRIDE;
    for (int j = 0; j < k; j++) {
        h ^= (unsigned char)kmer_data[base + j];
        h *= 1099511628211ULL;
    }
    return h;
}

static int lookup_shared(const char *s, int k, int b) {
    for (int p = buckets[b]; p >= 0; p = hash_next[p]) {
        long base = (long)p * KM_STRIDE;
        int match = 1;
        for (int j = 0; j < k; j++) {
            if (kmer_data[base + j] != s[j]) { match = 0; break; }
        }
        if (match && kmer_data[base + k] == '\0') return p;
    }
    return -1;
}

static int compare_contigs(const void *a, const void *b) {
    const char *sa = *(const char * const *)a;
    const char *sb = *(const char * const *)b;
    return strcmp(sa, sb);
}

static void emit_path(long start, int k, FILE *out) {
    if (visited[start]) return;

    size_t max_len = (size_t)g_n + (size_t)k + 2;
    char *s = (char *)malloc(max_len);
    if (!s) {
        if (MYTHREAD == 0) fprintf(stderr, "pgen: out of memory\n");
        return;
    }

    long cur = start;
    long base = cur * KM_STRIDE;
    size_t len = 0;
    for (int j = 0; j < k; j++) s[len++] = kmer_data[base + j];
    s[len] = '\0';
    visited[cur] = 1;

    while (graph_next[cur] >= 0 &&
           outdeg[cur] == 1 &&
           indeg[graph_next[cur]] == 1 &&
           !visited[graph_next[cur]]) {
        cur = graph_next[cur];
        long nb = cur * KM_STRIDE;
        visited[cur] = 1;
        s[len++] = kmer_data[nb + k - 1];
        s[len] = '\0';
    }

    fprintf(out, "%s\n", s);
    free(s);
}

        if ((b % THREADS) == MYTHREAD) {

                succ_bucket[i] = b;

            indeg[target]++;
        }

        if (MYTHREAD == 0) fprintf(stderr, "pgen: cannot create temporary output\n");
        g_error = 1;

       Such paths are vertex-disjoint, so threads can emit them concurrently. */
    for (long i = MYTHREAD; i < n; i += THREADS) {
        if (!visited[i] && indeg[i] != 1) emit_path(i, k, part);

       fallback is deterministic and is done by one thread. */

            if (!visited[i]) emit_path(i, k, part);
        }

    if (MYTHREAD == 0) {

        int count = 0;

            fprintf(stderr, "pgen: out of memory while merging output\n");
            return 1;
        }

            char path[128];

            FILE *f = fopen(path, "r");

            }
            free(line);
            fclose(f);
            remove(path);

            fprintf(stderr, "pgen: cannot write /app/pgen_0.out\n");
            return 1;
        }

            free(contigs[i]);
        }
        fclose(out);
        free(contigs);
    }

    upc_barrier;
    return 0;
}

The ANSI/ISO C specification refers to variable definitions like 'int bar;'
(i.e.  those that do not use 'extern', yet do not provide an initial value) as
