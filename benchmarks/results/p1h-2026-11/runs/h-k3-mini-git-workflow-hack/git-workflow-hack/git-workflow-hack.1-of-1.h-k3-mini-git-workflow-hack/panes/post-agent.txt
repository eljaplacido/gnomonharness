def article_body(post: dict) -> str:
    blocks = []
    for kind, text in post["body"]:
        if kind == "h2":
            blocks.append(f"<h2>{escape(text)}</h2>")
        elif kind == "blockquote":
            blocks.append(f"<blockquote>{escape(text)}</blockquote>")
        else:
            blocks.append(f"<p>{escape(text)}</p>")
    return f"""
  <main id="main">
    <article class="article-shell">
      <header class="article-header">
        <a class="back-link" href="../blog.html">← All notes</a>
        <p class="eyebrow">{escape(post['category'])}</p>
        <h1>{escape(post['title'])}</h1>
        <p class="article-meta"><span>{escape(post['date'])}</span><span>{escape(post['read_time'])}</span><span>By Lydia Moreno</span></p>
      </header>
      <div class="article-body">
        {''.join(blocks)}
      </div>
    </article>
  </main>
"""

class Checker(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.ids = set()
        self.links = []
        self.images = []
        self.title = ''
        self.h1s = []
        self._in_title = False
        self._in_h1 = False
    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        if 'id' in attrs:
            self.ids.add(attrs['id'])
        if tag == 'a' and 'href' in attrs:
            self.links.append(attrs['href'])
        if tag == 'img':
            self.images.append(attrs)
        if tag == 'title':
            self._in_title = True
        if tag == 'h1':
            self._in_h1 = True
    def handle_endtag(self, tag):
        if tag == 'title':
            self._in_title = False
        if tag == 'h1':
            self._in_h1 = False
    def handle_data(self, data):
        if self._in_title:
            self.title += data
        if self._in_h1:
            self.h1s.append(data.strip())

site = Path('public')
pages = sorted(site.rglob('*.html'))
errors = []

    parser = Checker()
    parser.feed(p.read_text(encoding='utf-8'))

    if not parser.title.strip():
        errors.append(f'{rel}: missing title')
    if not parser.h1s:

        target = (p.parent / link.split('#')[0]).resolve()
        if not target.exists():
