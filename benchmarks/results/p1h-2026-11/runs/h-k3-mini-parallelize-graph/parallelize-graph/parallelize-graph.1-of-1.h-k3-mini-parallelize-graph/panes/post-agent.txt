        if (b % THREADS == MYTHREAD) {
            nexts[i] = buckets[b];
            buckets[b] = i;
        }
    }
    upc_barrier;

    keybuf = (char *)malloc((size_t)k + 1);

        perror("malloc");
        return 1;

            size_t x;

            globfree(&gl);

        perror(outpath);
        return 1;

    /* Phase 5: traverse the maximal paths in parallel.  Roots are assigned
       to threads by k-mer index; a root has no predecessor in the graph. */
    for (i = MYTHREAD; i < n; i += THREADS) {
        if (!visited[i] &&
            predecessor_of(i, kmers, stride, k, lefts,
                           buckets, hsize, nexts, keybuf) < 0) {
            emit_path(out, i, 0, kmers, stride, k, lefts, rights,
                      buckets, hsize, nexts, visited, buf, keybuf);
        }

    free(keybuf);

    upc_free(lefts);
    upc_free(rights);
    upc_free(nexts);
    upc_free(visited);
    upc_free(buckets);
    return 0;
