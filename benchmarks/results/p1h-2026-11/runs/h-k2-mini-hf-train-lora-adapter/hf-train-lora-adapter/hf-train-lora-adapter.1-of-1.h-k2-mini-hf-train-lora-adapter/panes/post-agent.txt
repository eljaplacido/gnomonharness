#!/usr/bin/env python3
"""LoRA fine-tuning for biomedical multiple-choice QA with GPT-2.

The model is trained as a causal language model to produce the correct answer
letter after a standardized question/options prompt.  Only the answer tokens
(and EOS) contribute to the training loss.
"""
import argparse
import json
import math
import os
import random
import time
from pathlib import Path

import torch
from peft import LoraConfig, TaskType, get_peft_model
from transformers import AutoModelForCausalLM, AutoTokenizer

BASE_MODEL = "openai-community/gpt2"
DATA_DIR = Path("/app/data")
OUTPUT_DIR = Path("/app/experiments/lora_adapter")
RESULTS_PATH = Path("/app/lora_results.json")


def set_seed(seed: int) -> None:
    random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


def make_prompt(sample: dict) -> str:
    options = "\n".join(sample["options"])
    return f"Question: {sample['question']}\nOptions:\n{options}\nAnswer:"


def make_training_text(sample: dict) -> str:
    return make_prompt(sample) + f" {sample['answer']}"


def encode_example(tokenizer, sample: dict, max_length: int):
    """Tokenize one example and mask all prompt positions in the labels."""
    prompt = make_prompt(sample)
    text = prompt + f" {sample['answer']}" + tokenizer.eos_token
    encoded = tokenizer(text, truncation=True, max_length=max_length,
                        return_attention_mask=True)
    prompt_len = len(tokenizer(prompt, add_special_tokens=False)["input_ids"])
    labels = list(encoded["input_ids"])
    for i in range(min(prompt_len, len(labels))):
        labels[i] = -100
    return {
        "input_ids": encoded["input_ids"],
        "attention_mask": encoded["attention_mask"],
        "labels": labels,
    }


def collate(batch, pad_token_id: int):
    max_len = max(len(x["input_ids"]) for x in batch)
    input_ids, attention_mask, labels = [], [], []
    for x in batch:
        pad = max_len - len(x["input_ids"])
        # Left padding is generation friendly and harmless for training because
        # padded labels are ignored via -100.
        input_ids.append([pad_token_id] * pad + x["input_ids"])
        attention_mask.append([0] * pad + x["attention_mask"])
        labels.append([-100] * pad + x["labels"])
    return {
        "input_ids": torch.tensor(input_ids, dtype=torch.long),
        "attention_mask": torch.tensor(attention_mask, dtype=torch.long),
        "labels": torch.tensor(labels, dtype=torch.long),
    }


@torch.no_grad()
def evaluate_next_token(model, tokenizer, samples, max_length: int, batch_size: int):
    """Validate greedy first-token decoding of A/B/C/D."""
    model.eval()
    letter_ids = {letter: tokenizer(f" {letter}", add_special_tokens=False)["input_ids"][0]
                  for letter in "ABCD"}
    correct = 0
    total = 0
    for start in range(0, len(samples), batch_size):
        batch_samples = samples[start:start + batch_size]
        prompts = [make_prompt(s) for s in batch_samples]
        enc = tokenizer(prompts, padding=True, truncation=True,
                        max_length=max_length, return_tensors="pt")
        logits = model(**enc).logits
        lengths = enc["attention_mask"].sum(dim=1).tolist()
        for row, sample in enumerate(batch_samples):
            # With left padding, the final real token is always the last row position.
            next_logits = logits[row, -1]
            pred_id = max(letter_ids.values(), key=lambda tid: float(next_logits[tid]))
            pred = [k for k, v in letter_ids.items() if v == pred_id][0]
            correct += int(pred == sample["answer"])
            total += 1
    model.train()
    return correct / max(total, 1)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--epochs", type=int, default=8)
    parser.add_argument("--batch-size", type=int, default=16)
    parser.add_argument("--grad-accum", type=int, default=1)
    parser.add_argument("--lr", type=float, default=5e-4)
    parser.add_argument("--weight-decay", type=float, default=0.01)
    parser.add_argument("--max-length", type=int, default=256)
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--threads", type=int, default=20)
    parser.add_argument("--eval-batch-size", type=int, default=32)
    parser.add_argument("--output-dir", type=str, default=str(OUTPUT_DIR))
    parser.add_argument("--results-path", type=str, default=str(RESULTS_PATH))
    args = parser.parse_args()

    set_seed(args.seed)
    torch.set_num_threads(args.threads)
    output_dir = Path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    with open(DATA_DIR / "train.json", encoding="utf-8") as f:
        train_samples = json.load(f)
    with open(DATA_DIR / "val.json", encoding="utf-8") as f:
        val_samples = json.load(f)

    tokenizer = AutoTokenizer.from_pretrained(BASE_MODEL)
    tokenizer.pad_token = tokenizer.eos_token
    tokenizer.padding_side = "left"

    model = AutoModelForCausalLM.from_pretrained(BASE_MODEL)
    # Keep the base model static and train only the low-rank attention adapters.
    model.config.use_cache = False

    lora_config = LoraConfig(
        r=8,
        lora_alpha=16,
        target_modules=["c_attn"],
        lora_dropout=0.05,
        bias="none",
        task_type=TaskType.CAUSAL_LM,
        fan_in_fan_out=True,
    )
    model = get_peft_model(model, lora_config)
    model.print_trainable_parameters()

    train_enc = [encode_example(tokenizer, s, args.max_length) for s in train_samples]
    optimizer = torch.optim.AdamW(
        [p for p in model.parameters() if p.requires_grad],
        lr=args.lr,
        weight_decay=args.weight_decay,
    )
    # A short warmup plus cosine decay is stable for small LoRA runs.
    steps_per_epoch = math.ceil(len(train_enc) / (args.batch_size * args.grad_accum))
    total_steps = max(1, steps_per_epoch * args.epochs)
    warmup_steps = max(1, int(0.05 * total_steps))

    def lr_lambda(step):
        if step < warmup_steps:
            return step / warmup_steps
        progress = (step - warmup_steps) / max(1, total_steps - warmup_steps)
        return 0.05 + 0.95 * 0.5 * (1.0 + math.cos(math.pi * progress))

    scheduler = torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)

    best_acc = -1.0
    best_epoch = 0
    history = []
    global_step = 0
    for epoch in range(1, args.epochs + 1):
        model.train()
        order = list(range(len(train_enc)))
        random.Random(args.seed + epoch).shuffle(order)
        running_loss = 0.0
        n_losses = 0
        start_time = time.time()
        optimizer.zero_grad(set_to_none=True)

        for batch_idx, start in enumerate(range(0, len(order), args.batch_size)):
            ids = order[start:start + args.batch_size]
            batch = collate([train_enc[i] for i in ids], tokenizer.pad_token_id)
            out = model(**batch)
            loss = out.loss / args.grad_accum
            loss.backward()
            running_loss += float(out.loss.detach())
            n_losses += 1

            if (batch_idx + 1) % args.grad_accum == 0 or (start + args.batch_size) >= len(order):
                torch.nn.utils.clip_grad_norm_(
                    [p for p in model.parameters() if p.requires_grad], 1.0
                )
                optimizer.step()
                scheduler.step()
                optimizer.zero_grad(set_to_none=True)
                global_step += 1

        train_loss = running_loss / max(1, n_losses)
        val_acc = evaluate_next_token(
            model, tokenizer, val_samples, args.max_length, args.eval_batch_size
        )
        elapsed = time.time() - start_time
        history.append({"epoch": epoch, "train_loss": train_loss, "val_accuracy": val_acc})
        print(
            f"epoch {epoch}/{args.epochs} | loss {train_loss:.4f} | "
            f"val_acc {val_acc:.2%} | {elapsed:.1f}s",
            flush=True,
        )

        if val_acc > best_acc:
            best_acc = val_acc
            best_epoch = epoch
            model.save_pretrained(str(output_dir))
